function walkFiles(root, predicate, limit = 200) {
  const found = [];
  if (!exists(root)) return found;
  const stack = [root];
  while (stack.length && found.length < limit) {
    const current = stack.pop();
    let entries = [];
    try {
      entries = fs.readdirSync(current, { withFileTypes: true });
    } catch {
      continue;
    }
    for (const entry of entries) {
      const fullPath = path.join(current, entry.name);
      if (entry.isDirectory()) {
        if (!ignoredDirs.has(entry.name)) stack.push(fullPath);
        continue;
      }
      if (entry.isFile() && predicate(fullPath)) found.push(fullPath);
      if (found.length >= limit) break;
    }
  }
  return found;
}

function findCodeSnapshots(root) {

    path.join(root, "00_admin"),
    path.join(root, "99_local_pc_transferred"),

  const matches = [];
  for (const snapshotRoot of snapshotRoots) {

          return name.includes("snapshot") && [".zip", ".tar", ".tgz", ".gz"].includes(path.extname(name));

  }
  return matches;
}

function countFiles(root) {
  let count = 0;
  if (!exists(root)) return 0;
  const stack = [root];
  while (stack.length) {
    const current = stack.pop();
    let entries = [];
    try {
      entries = fs.readdirSync(current, { withFileTypes: true });
    } catch {
      continue;
    }
    for (const entry of entries) {
      const fullPath = path.join(current, entry.name);
      if (entry.isDirectory()) stack.push(fullPath);
      if (entry.isFile()) count += 1;
    }
  }
  return count;
}
