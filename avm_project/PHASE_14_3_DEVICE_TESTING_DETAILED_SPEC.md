```ruby
post_install do |installer|
  installer.pods_project.targets.each do |target|
    flutter_additional_ios_build_settings(target)
    target.build_configurations.each do |config|
      config.build_settings['GCC_PREPROCESSOR_DEFINITIONS'] ||= [
        '$(inherited)',
        'COCOAPODS=1',
      ]
    end
  end
end
```

```swift
// FeatureEngineeringTests.swift

class FeatureEngineeringTests: XCTestCase {
    
    // Test 1: 22개 특징 계약 검증
    func testContractHas22Features() {
        XCTAssertEqual(22, FeatureEngineering.featureOrder.size)
    }
    
    // Test 2: 벡터 길이 검증
    func testVectorMatchesContractLength() {
        let input = PropertyInput(
            region: "Seoul",
            areaM2: 84.0,
            yearBuilt: 2015
        )
        XCTAssertEqual(22, FeatureEngineering.buildVector(input).count)
    }
    
    // Test 3: 특징 순서 검증
    func testFeatureOrderMatchesTrainedColumns() {
        XCTAssertEqual("area_m2", FeatureEngineering.featureOrder[0])
        XCTAssertEqual("price_per_pyeong", FeatureEngineering.featureOrder[21])
        XCTAssertEqual("building_age", FeatureEngineering.featureOrder[4])
    }
    
    // Test 4: 평당 가격 계산
    func testPricePerPyeongDerivation() {
        let input = PropertyInput(
            region: "Seoul",
            areaM2: 84.0,
            yearBuilt: 2015
        )
        let features = FeatureEngineering.buildFeatures(input)
        XCTAssertEqual(84.0 / 3.3, features["price_per_pyeong"]!, accuracy: 1e-6)
    }
    
    // Test 5: 건물 나이 감가 계산
    func testAgeDepreciationBuckets() {
        XCTAssertEqual(1.0, FeatureEngineering.ageDepreciation(3.0), accuracy: 1e-9)
        XCTAssertEqual(0.95, FeatureEngineering.ageDepreciation(10.0), accuracy: 1e-9)
        XCTAssertEqual(0.88, FeatureEngineering.ageDepreciation(11.0), accuracy: 1e-9)
        XCTAssertEqual(0.78, FeatureEngineering.ageDepreciation(20.0), accuracy: 1e-9)
        XCTAssertEqual(0.65, FeatureEngineering.ageDepreciation(25.0), accuracy: 1e-9)
    }
    
    // Test 6: 지역 메타정보 적용
    func testRegionMetaApplied() {
        let input = PropertyInput(
            region: "Busan",
            areaM2: 84.0,
            yearBuilt: 2015
        )
        let features = FeatureEngineering.buildFeatures(input)
        XCTAssertEqual(1.04, features["brand_premium"]!, accuracy: 1e-6)
        XCTAssertEqual(37.301888, features["latitude"]!, accuracy: 1e-6)
    }
}
```
